cmake_minimum_required(VERSION 3.25)
project(WiiCompiledGraphicsTests LANGUAGES C CXX OBJC OBJCXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64" OR NOT CMAKE_OSX_SYSROOT MATCHES "iphoneos")
  message(FATAL_ERROR "Physical arm64 iphoneos SDK required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_GX ON CACHE BOOL "" FORCE)
set(AURORA_ENABLE_DVD OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_CARD OFF CACHE BOOL "" FORCE)
set(AURORA_DAWN_PROVIDER vendor CACHE STRING "" FORCE)
set(AURORA_SDL3_PROVIDER vendor CACHE STRING "" FORCE)
set(AURORA_DAWN_LINKAGE static CACHE STRING "" FORCE)
set(AURORA_SDL3_LINKAGE static CACHE STRING "" FORCE)
set(DAWN_ENABLE_METAL ON CACHE BOOL "" FORCE)
foreach(backend D3D11 D3D12 VULKAN DESKTOP_GL OPENGLES NULL)
  set(DAWN_ENABLE_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(../aurora-main aurora)
add_executable(WiiCompiledGraphicsTests MACOSX_BUNDLE main.cpp)
target_link_libraries(WiiCompiledGraphicsTests PRIVATE aurora::core aurora::gx aurora::vi)
set_target_properties(WiiCompiledGraphicsTests PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER org.local.wiicompiled.graphicstests
  MACOSX_BUNDLE_BUNDLE_NAME WiiCompiledGraphicsTests)
