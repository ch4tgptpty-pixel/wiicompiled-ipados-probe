cmake_minimum_required(VERSION 3.25)
project(WiiCompiledGraphicsTests LANGUAGES C CXX OBJC OBJCXX)
string(TOLOWER "${CMAKE_OSX_SYSROOT}" probe_sysroot)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64" OR NOT probe_sysroot MATCHES "iphoneos")
  message(FATAL_ERROR "Physical arm64 iphoneos SDK required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_PkgConfig TRUE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(DAWN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DAWN_BUILD_PROTOBUF OFF CACHE BOOL "" FORCE)
set(TINT_BUILD_IR_BINARY OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_GX ON CACHE BOOL "" FORCE)
set(AURORA_ENABLE_DVD OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_CARD OFF CACHE BOOL "" FORCE)
set(AURORA_DAWN_PROVIDER vendor CACHE STRING "" FORCE)
set(AURORA_SDL3_PROVIDER vendor CACHE STRING "" FORCE)
set(AURORA_DAWN_LINKAGE static CACHE STRING "" FORCE)
set(AURORA_SDL3_LINKAGE static CACHE STRING "" FORCE)
set(AURORA_DAWN_SOURCE_REVISION 13abc3bc8ea2d3c2050f9e77a12d012108ceee24 CACHE STRING "" FORCE)
set(AURORA_SDL3_SOURCE_REVISION 5848e584a1b606de26e3dbd1c7e4ecbc34f807a6 CACHE STRING "" FORCE)
set(DAWN_ENABLE_METAL ON CACHE BOOL "" FORCE)
foreach(backend D3D11 D3D12 VULKAN DESKTOP_GL OPENGLES NULL)
  set(DAWN_ENABLE_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(../aurora-main aurora)
target_compile_definitions(aurora_core PRIVATE AURORA_SDL_CALLBACKS)
target_compile_definitions(aurora_gx PRIVATE AURORA_SDL_CALLBACKS)
add_executable(WiiCompiledGraphicsTests MACOSX_BUNDLE main.cpp scene.cpp status.mm)
target_include_directories(WiiCompiledGraphicsTests PRIVATE ../aurora-main/lib)
target_link_libraries(WiiCompiledGraphicsTests PRIVATE aurora::core aurora::gx aurora::vi dawn::webgpu_dawn "-framework UIKit" "-framework Foundation")
set_source_files_properties(status.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
set_target_properties(WiiCompiledGraphicsTests PROPERTIES
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist"
  MACOSX_BUNDLE_GUI_IDENTIFIER org.local.wiicompiled.graphicstests
  MACOSX_BUNDLE_BUNDLE_NAME WiiCompiledGraphicsTests)
